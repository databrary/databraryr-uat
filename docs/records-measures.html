About

This page documents testing of the functions that relate to records and measures.

Important

Most of the examples below were created by Pawel Armatys from Montrose Software.

this_endpoint <- databraryr:::DATABRARY_BASE_URL

if (params$test_endpoint == "production") {
  message("Test target: Production API.")
  if (this_endpoint == params$prod_api) {
    message("Databraryr API target endpoint is correct.")
  } else {
    message(paste0(
      "Target: ",
      params$test_endpoint,
      " | databraryr base: ",
      this_endpoint
    ))
    stop(
      "`databraryr` API target endpoint does not match testing target endpoint. Check ~/.Renviron."
    )
  }
} else if (params$test_endpoint == "staging") {
  message("Test target: Staging API.")
  if (this_endpoint == params$staging_api) {
    message("Databraryr API target endpoint is correct.")
  } else {
    message(paste0(
      "Target: ",
      params$test_endpoint,
      " | databraryr base: ",
      this_endpoint
    ))
    stop(
      "`databraryr` API target endpoint does not match testing target endpoint. Check ~/.Renviron."
    )
  }
} else {
  stop("API endpoint not specified properly")
}
